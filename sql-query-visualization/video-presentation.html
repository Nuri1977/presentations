-- Top revenue customers this month
SELECT
  customer_id,
  SUM(order_total) AS revenue,
  COUNT(*) AS orders
FROM mart.fct_orders
WHERE order_date >= '2026-01-01'
GROUP BY 1
ORDER BY revenue DESC
LIMIT 10;
