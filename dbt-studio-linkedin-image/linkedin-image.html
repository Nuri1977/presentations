"Show monthly revenue by customer segment — only paying customers with at least one order in the last 12 months."
✓ Generated in 4s
monthly_revenue_by_segment.sql New
with paying_customers as (
  select customer_id
  from {{ ref('orders') }}
  where status = 'completed'
  and order_date >= dateadd('month',-12,current_date())
  group by 1
)
select
  date_trunc('month', o.order_date) as order_month,
  c.segment as segment,
  sum(o.amount) as total_revenue
from {{ ref('orders') }} o
join {{ ref('customers') }} c using(customer_id)
join paying_customers pc using(customer_id)
group by 1, 2
